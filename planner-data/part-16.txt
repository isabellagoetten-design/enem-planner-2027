function ensureRoadmapContent(subject,category,topic){
 if(!contentCatalog[subject])contentCatalog[subject]={};
 const target=normalizeTopicText(topic);
 for(const [existingCategory,items] of Object.entries(contentCatalog[subject])){
  const found=items.find(item=>normalizeTopicText(item)===target);
  if(found)return {category:existingCategory,topic:found};
 }
 if(!contentCatalog[subject][category])contentCatalog[subject][category]=[];
 contentCatalog[subject][category].push(topic);
 return {category,topic};
}
